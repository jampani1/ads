import { Text, View, FlatList } from 'react-native';

export default function App() {
const dados = [
  {id: '1', nome: 'João'},
  {id: '2', nome: 'Carlos'},
  {id: '3', nome: 'Emanuel'}
];

  return (
    <View>
      <FlatList
        data = {dados}
        keyExtractor = {(dados) =>dados.id}
        renderItem = {({item}) =><Text>{item.nome
        }</Text>}
      />
    </View>
  )
}

  }

      <Pessoa nome="maria"/>
    </View>
